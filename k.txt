public void dda(Graphics g, int x0, int y0, int x1, int y1) {

    // Calculate differences in x and y
    int dx = x1 - x0;
    int dy = y1 - y0;

    // Calculate total steps required to plot the line
    int steps = Math.max(Math.abs(dx), Math.abs(dy));

    // Calculate incremental values for x and y
    float xInc = (float) dx / steps;
    float yInc = (float) dy / steps;

    // Set starting coordinate
    float x = x0;
    float y = y0;

    // Loop through steps and draw pixel simulation using fillRect
    for (int i = 0; i <= steps; i++) {
        g.fillRect(Math.round(x), Math.round(y), 2, 2);
        x += xInc;
        y += yInc;
    }
}

    public void drawBresenhamLine(Graphics g, int x1, int y1, int x2, int y2) {
        int dx = Math.abs(x2 - x1);
        int dy = Math.abs(y2 - y1);

        int sx = (x2 > x1) ? 1 : -1;
        int sy = (y2 > y1) ? 1 : -1;

        if (dx >= dy) {
            int d = 2 * dy - dx;
            int y = y1;

            for (int x = x1; x != x2; x += sx) {
                g.fillRect(x, y, 2, 2);

                if (d > 0) {
                    y += sy;
                    d -= 2 * dx;        
                }

                d += 2 * dy;
            }

        } else {
            int d = 2 * dx - dy;
            int x = x1;

            for (int y = y1; y != y2; y += sy) {
                g.fillRect(x, y, 2, 2);

                if (d > 0) {
                    x += sx;
                    d -= 2 * dy;
                }

                d += 2 * dx;
            }
        }
    }

// (b) Bresenham's t Circle Drawing Algorithm Implementation

public void BreCir(Graphics g, int xc, int yc, int r) {
    int x = 0;
    int y = r;
    int d = 3 - 2 * r;

    // Plot initial symmetric points in all 8 octants
    plotPoints(g, xc, yc, x, y);

    // Loop until y < x
    while (y >= x) {
        x++;

        if (d > 0) {
            y--;
            d = d + 4 * (x - y) + 10;
        } else {
            d = d + 4 * x + 6;
        }

        plotPoints(g, xc, yc, x, y);
    }
}

// Helper method implementing 8-way symmetry plotting
private void plotPoints(Graphics g, int xc, int yc, int x, int y) {
    g.fillRect(xc + x, yc + y, 2, 2);
    g.fillRect(xc - x, yc + y, 2, 2);
    g.fillRect(xc + x, yc - y, 2, 2);
    g.fillRect(xc - x, yc - y, 2, 2);

    g.fillRect(xc + y, yc + x, 2, 2);
    g.fillRect(xc - y, yc + x, 2, 2);
    g.fillRect(xc + y, yc - x, 2, 2);
    g.fillRect(xc - y, yc - x, 2, 2);
}

// (b) Midpoint Circle Drawing Algorithm Method (BreCir)

public void BreCir(Graphics g, int xc, int yc, int r) {
    int x = 0;
    int y = r;
    int p = 1 - r; // Exact Midpoint Decision Parameter

    plotPoints(g, xc, yc, x, y);

    while (x < y) {
        x++;

        if (p < 0) {
            p += 2 * x + 1; // p < 0 වූ විට
        } else {
            y--;
            p += 2 * (x - y) + 1; // p >= 0 වූ විට
        }

        plotPoints(g, xc, yc, x, y);
    }
}

		float x = x1, y = y1;
		for (int i = 0; i <= steps; i++) {
			g.fillRect(Math.round(x), Math.round(y), 2, 2);
			x += xInc;
			y += yInc;
		}
	}

	void drawCircle(Graphics g, int xc, int yc, int r) {
		int x = 0, y = r, d = 1 - r;
		while (x <= y) {
			plot(g, xc, yc, x, y);
			if (d < 0) d += 2 * x + 1;
			else { d += 2 * (x - y) + 1; y--; }
			x++;
		}
	}

	void plot(Graphics g, int xc, int yc, int x, int y) {
		g.fillRect(xc + x, yc + y, 2, 2);  g.fillRect(xc - x, yc + y, 2, 2);

		g.fillRect(xc + y, yc - x, 2, 2);  g.fillRect(xc - y, yc - x, 2, 2);
	}

	// Smile = only the lower half of a midpoint circle

			else { d += 2 * (x - y) + 1; y--; }
			x++;
		}
	}

package paper21;

import javax.swing.*;
import java.awt.*;

public class HouseTransformation extends JPanel {

    // DDA Line Drawing Algorithm
    public void dda(Graphics g, int x0, int y0, int x1, int y1) {
        int dx = x1 - x0;
        int dy = y1 - y0;

        int steps = Math.max(Math.abs(dx), Math.abs(dy));

        if (steps == 0) {
            g.fillRect(x0, y0, 2, 2);
            return;
        }

        float xInc = (float) dx / steps;
        float yInc = (float) dy / steps;

        float x = x0;
        float y = y0;

        for (int i = 0; i <= steps; i++) {
            g.fillRect(Math.round(x), Math.round(y), 2, 2);
            x += xInc;
            y += yInc;
        }
    }

    // Draw a shape using DDA
    private void drawShape(int[][] shape, Graphics g) {
        for (int i = 0; i < shape.length - 1; i++) {
            dda(g, shape[i][0], shape[i][1],
                shape[i + 1][0], shape[i + 1][1]);
        }
    }

    // (a) 2D Translation
    public void translate(int[][] xy, int tx, int ty) {
        for (int i = 0; i < xy.length; i++) {
            xy[i][0] += tx;
            xy[i][1] += ty;
        }
    }

    // (b) 2D Scaling
    public void scale(int[][] xy, int sx, int sy) {
        for (int i = 0; i < xy.length; i++) {
            xy[i][0] *= sx;
            xy[i][1] *= sy;
        }
    }

    // (c) 2D Rotation
    public void rotate(int[][] xy, double angle) {
        double rad = Math.toRadians(angle);
        double cos = Math.cos(rad);
        double sin = Math.sin(rad);

        for (int i = 0; i < xy.length; i++) {
            int x = xy[i][0];
            int y = xy[i][1];

            xy[i][0] = (int) Math.round(x * cos - y * sin);
            xy[i][1] = (int) Math.round(x * sin + y * cos);
        }
    }

    // Copy shape coordinates
    private int[][] copyShape(int[][] shape) {
        int[][] copy = new int[shape.length][2];

        for (int i = 0; i < shape.length; i++) {
            copy[i][0] = shape[i][0];
            copy[i][1] = shape[i][1];
        }

        return copy;
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);

        // Original house coordinates
        int[][] base = {
            {50, 150}, {150, 150}, {150, 250},
            {50, 250}, {50, 150}
        };

        int[][] roof = {
            {50, 150}, {100, 80},
            {150, 150}, {50, 150}
        };

        int[][] door = {
            {85, 250}, {115, 250},
            {115, 200}, {85, 200}, {85, 250}
        };

        // Draw original house in blue
        g.setColor(Color.BLUE);
        drawShape(base, g);
        drawShape(roof, g);
        drawShape(door, g);

        // Create copies of the house
        int[][] base2 = copyShape(base);
        int[][] roof2 = copyShape(roof);
        int[][] door2 = copyShape(door);

        // (i) Scale by 2 times
        scale(base2, 2, 2);
        scale(roof2, 2, 2);
        scale(door2, 2, 2);

        // (ii) Rotate by 90 degrees
        rotate(base2, 90);
        rotate(roof2, 90);
        rotate(door2, 90);

        // (iii) Translate to the right
        translate(base2, 400, 100);
        translate(roof2, 400, 100);
        translate(door2, 400, 100);

        // Draw transformed house in red
        g.setColor(Color.RED);
        drawShape(base2, g);
        drawShape(roof2, g);
        drawShape(door2, g);
    }

    public static void main(String[] args) {
        JFrame frame = new JFrame(
            "Question 02 - House Transformation"
        );

        HouseTransformation panel = new HouseTransformation();
        frame.add(panel);

        frame.setSize(1000, 700);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setLocationRelativeTo(null);
        frame.setVisible(true);
    }
}

public void rotate(int[][] xy, double angle) {
    // Convert angle from degrees to radians
    double rad = Math.toRadians(angle);
    double cos = Math.cos(rad);
    double sin = Math.sin(rad);

    for (int i = 0; i < xy.length; i++) {
        int x = xy[i][0];
        int y = xy[i][1];

        // Apply 2D rotation formulas:
        // X' = X * cos(θ) - Y * sin(θ)
        // Y' = X * sin(θ) + Y * cos(θ)

        xy[i][0] = (int) Math.round(x * cos - y * sin);
        xy[i][1] = (int) Math.round(x * sin + y * cos);
    }
}

    static final int[][] ROOF = {{-100, 100}, {0, 0}, {100, 100}, {-100, 100}};

    static void bresenham(Graphics g, int x0, int y0, int x1, int y1) {

        int dy = Math.abs(y1 - y0);
        int sx = (x0 < x1) ? 1 : -1;
        int sy = (y0 < y1) ? 1 : -1;

import javax.swing.*;
import java.awt.*;

public class HouseDrawing extends JPanel {

    
    static final int[][] BASE = {{-100, 100}, {100, 100}, {100, 300}, {-100, 300}, {-100, 100}};
    static final int[][] ROOF = {{-100, 100}, {0, 0}, {100, 100}, {-100, 100}};
    static final int[][] DOOR = {{-30, 300}, {30, 300}, {30, 220}, {-30, 220}, {-30, 300}};

    static final int CX = 400;   
    static final int CY = 400;   
    static final double SCALE = 0.6;

    public HouseDrawing() {
        setPreferredSize(new Dimension(800, 800));
        setBackground(new Color(238, 238, 238));
    }

    
    static int[][] transform(int[][] pts, double scale, double angleDeg, int tx, int ty) {
        double rad = Math.toRadians(angleDeg);
        double cos = Math.cos(rad);
        double sin = Math.sin(rad);
        int[][] out = new int[pts.length][2];
        for (int i = 0; i < pts.length; i++) {
            double x = pts[i][0] * scale;          
            double y = pts[i][1] * scale;
            double xr = x * cos - y * sin;         
            double yr = x * sin + y * cos;
            out[i][0] = (int) Math.round(xr) + tx; 
            out[i][1] = (int) Math.round(yr) + ty;
        }
        return out;
    }

   
    static void bresenham(Graphics g, int x0, int y0, int x1, int y1) {
        int dx = Math.abs(x1 - x0);
        int dy = Math.abs(y1 - y0);
        int sx = (x0 < x1) ? 1 : -1;
        int sy = (y0 < y1) ? 1 : -1;
        int err = dx - dy;

        while (true) {
            g.fillRect(x0, y0, 2, 2);              
            if (x0 == x1 && y0 == y1) break;
            int e2 = 2 * err;
            if (e2 > -dy) { err -= dy; x0 += sx; }
            if (e2 < dx)  { err += dx; y0 += sy; }
        }
    }

    
    static void drawShape(Graphics g, int[][] p) {
        for (int i = 0; i < p.length - 1; i++) {
            bresenham(g, p[i][0], p[i][1], p[i + 1][0], p[i + 1][1]);
        }
    }

   
    static void drawHouse(Graphics g, Color c, double angle, int tx, int ty) {
        g.setColor(c);
        drawShape(g, transform(BASE, SCALE, angle, tx, ty));
        drawShape(g, transform(ROOF, SCALE, angle, tx, ty));
        drawShape(g, transform(DOOR, SCALE, angle, tx, ty));
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        int d = 250;   

        drawHouse(g, Color.RED,                 0,   CX,     CY - d);  
        drawHouse(g, Color.GREEN,               90,  CX + d, CY);      
        drawHouse(g, new Color(255, 192, 0),    180, CX,     CY + d);  
        drawHouse(g, Color.MAGENTA,             270, CX - d, CY);      
    }

    public static void main(String[] args) {

package paper19;

public class midpointCirclecordinates {

    // (c) Bresenham's midpoint circle drawing algorithm
    public static void BreCir(int xc, int yc, int r) {
        int x = 0;
        int y = r;
        int p = 1 - r;                    // initial decision parameter

        plotPoints(xc, yc, x, y);

        while (x < y) {
            x++;
            if (p < 0) {
                p = p + 2 * x + 1;        // midpoint inside circle: keep y
            } else {
                y--;
                p = p + 2 * (x - y) + 1;  // midpoint outside circle: decrease y
            }
            plotPoints(xc, yc, x, y);
        }
    }

    // 8-way symmetry (avoids printing duplicate points)

        System.out.println("(" + (xc - y) + ", " + (yc + x) + ")");

void translateLine(Graphics g, int P[][], int T[]) {
    // P[0] = {x1, y1}, P[1] = {x2, y2}
    P[0][0] = P[0][0] + T[0];   // x1' = x1 + tx
    P[0][1] = P[0][1] + T[1];   // y1' = y1 + ty
    P[1][0] = P[1][0] + T[0];   // x2' = x2 + tx
    P[1][1] = P[1][1] + T[1];   // y2' = y2 + ty

    dda(g, P[0][0], P[0][1], P[1][0], P[1][1]);   // draw translated line
}

void scale(Graphics g, int x[], int y[], int sx, int sy) {
    int n = x.length;
    int nx[] = new int[n];
    int ny[] = new int[n];

    for (int i = 0; i < n; i++) {
        nx[i] = x[i] * sx;       // x' = x * sx
        ny[i] = y[i] * sy;       // y' = y * sy
    }

    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;     // next vertex (last joins first)
        dda(g, nx[i], ny[i], nx[j], ny[j]);
    }
}

void rotate(Graphics g, float a[][], int n, int x_pivot, int y_pivot, int angle) {
    double rad = Math.toRadians(angle);          // degrees -> radians
    double cos = Math.cos(rad);
    double sin = Math.sin(rad);

    int rx[] = new int[n];
    int ry[] = new int[n];

    for (int i = 0; i < n; i++) {
        float dx = a[i][0] - x_pivot;            // move pivot to origin
        float dy = a[i][1] - y_pivot;

        // rotate, then move back
        rx[i] = Math.round(x_pivot + dx * (float) cos - dy * (float) sin);
        ry[i] = Math.round(y_pivot + dx * (float) sin + dy * (float) cos);
    }

    // draw the rotated object using DDA
    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;                     // next vertex
        dda(g, rx[i], ry[i], rx[j], ry[j]);
    }
}

import javax.swing.*;

public class TriangleTransform extends JPanel {

	// ---------- DDA line ----------
	void dda(Graphics g, int x1, int y1, int x2, int y2) {
		int dx = x2 - x1, dy = y2 - y1;
		int steps = Math.max(Math.abs(dx), Math.abs(dy));
		float xInc = (float) dx / steps, yInc = (float) dy / steps;
		float x = x1, y = y1;
		for (int i = 0; i <= steps; i++) {
			g.fillRect(Math.round(x), Math.round(y), 2, 2);
			x += xInc;
			y += yInc;
		}
	}

	// ---------- draw a triangle with 3 DDA lines ----------
	void drawTriangle(Graphics g, int x[], int y[]) {
		for (int i = 0; i < 3; i++) {
			int j = (i + 1) % 3;                       // next vertex
			dda(g, x[i], y[i], x[j], y[j]);
		}
	}

	// ---------- Translate ----------
	void translate(Graphics g, int x[], int y[], int tx, int ty) {
		int nx[] = new int[3], ny[] = new int[3];
		for (int i = 0; i < 3; i++) {
			nx[i] = x[i] + tx;
			ny[i] = y[i] + ty;
		}
		drawTriangle(g, nx, ny);
	}

	// ---------- Scale (multiply each vertex) ----------
	void scale(Graphics g, int x[], int y[], int sx, int sy) {
		int nx[] = new int[3], ny[] = new int[3];
		for (int i = 0; i < 3; i++) {
			nx[i] = x[i] * sx;
			ny[i] = y[i] * sy;
		}
		drawTriangle(g, nx, ny);
	}

	// ---------- Rotate about a pivot (angle in degrees) ----------
	void rotate(Graphics g, float a[][], int n, int x_pivot, int y_pivot, int angle) {
		double rad = Math.toRadians(angle);
		double cos = Math.cos(rad), sin = Math.sin(rad);
		int rx[] = new int[n], ry[] = new int[n];
		for (int i = 0; i < n; i++) {
			float dx = a[i][0] - x_pivot, dy = a[i][1] - y_pivot;
			rx[i] = (int) Math.round(x_pivot + dx * cos - dy * sin);
			ry[i] = (int) Math.round(y_pivot + dx * sin + dy * cos);
		}
		drawTriangle(g, rx, ry);
	}

void rotate(Graphics g, float a[][], int n, int x_pivot, int y_pivot, int angle) {
    double rad = Math.toRadians(angle);
    double c = Math.cos(rad);
    double s = Math.sin(rad);

    for (int i = 0; i < n; i++) {
        float x = a[i][0] - x_pivot;          // shift pivot to origin
        float y = a[i][1] - y_pivot;

        a[i][0] = (float) (x * c - y * s) + x_pivot;   // rotate, shift back
        a[i][1] = (float) (x * s + y * c) + y_pivot;
    }

    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;
        dda(g, Math.round(a[i][0]), Math.round(a[i][1]),
               Math.round(a[j][0]), Math.round(a[j][1]));

import java.util.Scanner;

public class UserInputBresenhamLine {

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 1. Starting coordinate
        System.out.print("Enter starting x1: ");
        int x1 = sc.nextInt();
        System.out.print("Enter starting y1: ");
        int y1 = sc.nextInt();

        // 2. Ending coordinate
        System.out.print("Enter ending x2: ");
        int x2 = sc.nextInt();
        System.out.print("Enter ending y2: ");
        int y2 = sc.nextInt();

        // 3 & 4. Apply Bresenham and display all pixels
        System.out.println("\nPixels from (" + x1 + ", " + y1 + ") to (" + x2 + ", " + y2 + "):");
        bresenham(x1, y1, x2, y2);

        sc.close();
    }

    // Bresenham's line drawing algorithm (works for all slopes and directions)
    static void bresenham(int x1, int y1, int x2, int y2) {
        int dx = Math.abs(x2 - x1);
        int dy = Math.abs(y2 - y1);

        int sx = (x1 < x2) ? 1 : -1;   // step direction in x
        int sy = (y1 < y2) ? 1 : -1;   // step direction in y

        int x = x1;
        int y = y1;

        boolean steep = dy > dx;       // slope > 1: step along y instead of x
        if (steep) {
            int t = dx; dx = dy; dy = t;   // swap dx and dy
        }

        // Initial decision parameter
        int p = 2 * dy - dx;

        int count = 1;
        System.out.println(count + ": (" + x + ", " + y + ")");

        for (int i = 0; i < dx; i++) {
            if (p >= 0) {
                // move in both directions
                x += sx;
                y += sy;
                p = p + 2 * dy - 2 * dx;
            } else {
                // move only in the main direction
                if (steep) {
                    y += sy;
                } else {
                    x += sx;
                }
                p = p + 2 * dy;
            }
            count++;
            System.out.println(count + ": (" + x + ", " + y + ")");
        }
    }
}

        midpointCircle(window2[0], window2[1], window2[2], g);
        midpointCircle(doorWindow[0], doorWindow[1], doorWindow[2], g);
    }

    // Draws a polygon by connecting consecutive points with DDA lines
    private void drawPolygon(int[][] points, Graphics g) {
        for (int i = 0; i < points.length - 1; i++) {
            ddaLine(points[i][0], points[i][1], points[i + 1][0], points[i + 1][1], g);
        }
    }

    // ---------- DDA Line Drawing Algorithm ----------
    private void ddaLine(int x1, int y1, int x2, int y2, Graphics g) {
        int dx = x2 - x1;
        int dy = y2 - y1;
        int steps = Math.max(Math.abs(dx), Math.abs(dy));
        if (steps == 0) steps = 1;

        float Xinc = (float) dx / steps;
        float Yinc = (float) dy / steps;

        float x = x1;
        float y = y1;

        for (int i = 0; i <= steps; i++) {
            g.fillRect(Math.round(x), Math.round(y), 2, 2);
            x += Xinc;
            y += Yinc;
        }
    }

    // ---------- Midpoint Circle Drawing Algorithm ----------
    private void midpointCircle(int xc, int yc, int r, Graphics g) {
        int x = 0;
        int y = r;
        int d = 1 - r; // initial decision parameter

        plotCirclePoints(xc, yc, x, y, g);

        while (x < y) {
            x++;
            if (d < 0) {
                d += 2 * x + 1;
            } else {
                y--;
                d += 2 * (x - y) + 1;
            }
            plotCirclePoints(xc, yc, x, y, g);
        }
    }

    // Plots the 8 symmetric points of a circle
    private void plotCirclePoints(int xc, int yc, int x, int y, Graphics g) {
        g.fillRect(xc + x, yc + y, 2, 2);
        g.fillRect(xc - x, yc + y, 2, 2);
        g.fillRect(xc + x, yc - y, 2, 2);
        g.fillRect(xc - x, yc - y, 2, 2);
        g.fillRect(xc + y, yc + x, 2, 2);
        g.fillRect(xc - y, yc + x, 2, 2);
        g.fillRect(xc + y, yc - x, 2, 2);
        g.fillRect(xc - y, yc - x, 2, 2);
    }

    // ---------- Translation for line-based shapes ----------
    private void translate(int[][] points, int tx, int ty) {
        for (int i = 0; i < points.length; i++) {
            points[i][0] += tx;
            points[i][1] += ty;
        }
    }

    // ---------- Translation for circles {x, y, r} ----------
    private void translateCircle(int[] circle, int tx, int ty) {
        circle[0] += tx;
        circle[1] += ty;
        // radius stays the same
    }

    public static void main(String[] args) {
        JFrame frame = new JFrame("Task 4 - House Translation (Lines + Circles)");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(700, 500);
        frame.add(new Task4_HouseWithWindows());
        frame.setVisible(true);
    }
}

    // ---------- Midpoint Circle Algorithm (from the lab sheet) ----------
    static void drawCircle(Graphics g, int xc, int yc, int r) {
        int x = 0, y = r;
        int p = 1 - r;                       // initial decision parameter
        plotPoints(g, xc, yc, x, y);
        while (x < y) {
            x++;
            if (p < 0) {
                p += 2 * x + 1;
            } else {
                y--;
                p += 2 * (x - y) + 1;
            }
            plotPoints(g, xc, yc, x, y);
        }
    }

    // 8-way symmetry
    static void plotPoints(Graphics g, int xc, int yc, int x, int y) {
        g.fillRect(xc + x, yc + y, 2, 2);
        g.fillRect(xc - x, yc + y, 2, 2);
        g.fillRect(xc + x, yc - y, 2, 2);
        g.fillRect(xc - x, yc - y, 2, 2);
        g.fillRect(xc + y, yc + x, 2, 2);
        g.fillRect(xc - y, yc + x, 2, 2);
        g.fillRect(xc + y, yc - x, 2, 2);
        g.fillRect(xc - y, yc - x, 2, 2);
    }

import java.awt.*;
import javax.swing.*;

// This class extends JPanel to allow custom graphics rendering
public class DDALineDrawing extends JPanel {

    // The paint() method is called automatically when the window is rendered
    @Override
    public void paint(Graphics g) {
        super.paint(g);

        int x1 = 50, y1 = 50;
        int x2 = 300, y2 = 200;

        // Start and end points
        drawDDA(g, x1, y1, x2, y2);
    }

    // This method implements the DDA algorithm to draw a line
    public void drawDDA(Graphics g, int x1, int y1, int x2, int y2) {

        int dx = x2 - x1;   // Difference in x
        int dy = y2 - y1;   // Difference in y

        // Calculate steps
        int steps = Math.max(Math.abs(dx), Math.abs(dy));

        // Compute incremental values
        float x_inc = (float) dx / steps;
        float y_inc = (float) dy / steps;

        float x = x1;
        float y = y1;

        // Loop through steps to generate pixel points
        for (int i = 0; i <= steps; i++) {

            // Draw pixel as a small rectangle
            g.fillRect(Math.round(x), Math.round(y), 2, 2);

            x += x_inc;
            y += y_inc;
        }
    }

    // Main method to create and display the GUI
    public static void main(String[] args) {

        // Create a window frame
        JFrame frame = new JFrame("DDA Line Drawing Algorithm");

        DDALineDrawing panel = new DDALineDrawing();

        frame.add(panel);
        frame.setSize(400, 300);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setVisible(true);
    }
}

import java.awt.*;
import javax.swing.*;

public class BresenhamLineDrawing extends JPanel {

    @Override
    public void paint(Graphics g) {
        super.paint(g);
        drawBresenhamLine(g, 50, 100, 200, 150);
    }

    public void drawBresenhamLine(Graphics g, int x1, int y1, int x2, int y2) {

        int dx = x2 - x1;
        int dy = y2 - y1;

        int x = x1;
        int y = y1;

        // Initial decision parameter
        int p = 2 * dy - dx;

        // Plot first pixel
        g.fillRect(x, y, 2, 2);

        for (int i = 0; i < dx; i++) {

            x++;

            if (p < 0) {
                p = p + 2 * dy;
            } else {
                y++;
                p = p + 2 * dy - 2 * dx;
            }

            // Plot next pixel
            g.fillRect(x, y, 2, 2);
        }
    }

    public static void main(String[] args) {

        drawBresenhamLine(g, 200, 200, 150, 50);
    }

    public void drawBresenhamLine(Graphics g, int x1, int y1, int x2, int y2) {

        int dx = Math.abs(x2 - x1);
        int dy = Math.abs(y2 - y1);

        int sx = (x2 > x1) ? 1 : -1;
        int sy = (y2 > y1) ? 1 : -1;

        if (dx >= dy) {

            int d = 2 * dy - dx;
            int y = y1;

            for (int x = x1; x != x2; x += sx) {
                g.fillRect(x, y, 2, 2);

                if (d > 0) {
                    y += sy;
                    d -= 2 * dx;
                }

                d += 2 * dy;
            }

        } else {

            int d = 2 * dx - dy;
            int x = x1;

            for (int y = y1; y != y2; y += sy) {
                g.fillRect(x, y, 2, 2);

                if (d > 0) {
                    x += sx;
                    d -= 2 * dy;
                }

                d += 2 * dx;
            }
        }
    }
