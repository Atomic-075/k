import java.util.Stack;
public class InsertionSortStack{
	int stk[] = new int[5];
	int top = -1;
	int size = stk.length;
	
	public void push(int elt){
		if(top!= size){
			top++;
			stk[top] = elt;
			
			System.out.println(elt + "     is Inserted");
		}
		else{
			System.out.println(" Stack is full");
		}		
	}
	public void sort(){
		for(int i=0; i<=top;i++){
			int minIdx = i;
			for(int j=i+1;j<=top;j++){
				if(stk[j]< stk[minIdx]){
					minIdx = j;
				}
			}
			int temp = stk[minIdx];
			stk[minIdx] = stk[i];
			stk[i] = temp;
		}	
		System.out.println( "After  Sort: ");	
		for( int i=0;i<=top;i++){
			System.out.println( stk[i] + "   ");
		}
		//System.out.println();
	}
}

class mainstk{
		public static void main(String[]args){
			InsertionSortStack sk = new InsertionSortStack();
				
				sk.push(5);
				sk.push(1);
				sk.push(3);
				sk.push(2);
				sk.sort();
		}
}






class binarySearch{
	public static void main(String[]args){
		int[]A = {2,4,6,8,10};
		int l=0, h=A.length-1, x=3;
		boolean found = false;
		while(l<=h){
			int mid = (l + h)/2;
			if(A[mid] == x){
				System.out.println("Target found at index : " + mid);
				found = true;
				break;
			}
			else if(x>A[mid]){
				l = mid+1;
			}
			else{
				h = mid-1;
			}
		}
		if(!found){
			System.out.println("Target is not in the Array.");
		}
	}
}
